Table posts {
  id uuid [primary key]
  title varchar [not null]
  preview varchar(200) [note: 'Preview of the post']
  url text [not null]
  published_at timestamp [not null]
  ingested_at timestamp [not null]
  image text 
  tags string [not null]
  source_id string [not null]
}

Table tags {
  id uuid [primary key]
  name varchar unique [not null]
  posts text [not null]
}

Table sources {
  id varchar [primary key]
  name varchar [not null]
}

Table metrics {
  source_id id [primary key]
  post_amount integer
}


Table post_tags {
  post_id uuid [not null, ref: > posts.id]
  tag_id integer [not null, ref: > tags.id]

  indexes {
    (post_id, tag_id) [pk]
    (tag_id, post_id)
  }
}

Ref post_source: posts.source_id > sources.id
